{{- if eq .Values.inference.framework "triton-vllm" }}
Triton-VLLM Service:
  $ kubectl get svc {{ .Values.inference.serviceName }} -n {{ .Values.inference.serviceNamespace }}

  This deployment runs NVIDIA Triton Inference Server with a vLLM backend.
  It exposes the Triton API, NOT the OpenAI API:

  Ports:
    HTTP:    {{ .Values.inference.serviceName }}:8000
    gRPC:    {{ .Values.inference.serviceName }}:8001
    Metrics: {{ .Values.inference.serviceName }}:8002

  List loaded models:
  $ curl -X POST http://{{ .Values.inference.serviceName }}:8000/v2/repository/index
  $ curl http://{{ .Values.inference.serviceName }}:8000/v2/models/vllm_model

  Check server readiness (waits for model to be READY):
  $ curl http://{{ .Values.inference.serviceName }}:8000/v2/health/ready

  Example inference request (gRPC streaming on port 8001 — required because the
  vLLM backend declares the model as decoupled; plain HTTP/gRPC infer is NOT
  supported by this backend):
  pip3 install 'tritonclient[grpc]'

  python3 - <<'EOF'
  import time
  import numpy as np
  import tritonclient.grpc as grpcclient

  client = grpcclient.InferenceServerClient(url="{{ .Values.inference.serviceName }}:8001")

  def on_result(result, error):
      if error:
          print("ERR:", error)
          return
      out = result.as_numpy("text_output")
      if out is not None:
          for row in out:
              print("OUT:", row.decode() if isinstance(row, bytes) else row)

  client.start_stream(on_result)
  inp = grpcclient.InferInput("text_input", [1], "BYTES")
  inp.set_data_from_numpy(np.array(["Hello, how are you?"], dtype=np.object_))
  stream_in = grpcclient.InferInput("stream", [1], "BOOL")
  stream_in.set_data_from_numpy(np.array([True], dtype=bool))
  samp_in = grpcclient.InferInput("sampling_parameters", [1], "BYTES")
  samp_in.set_data_from_numpy(np.array(['{"temperature":0.7,"max_tokens":50}'], dtype=np.object_))
  client.async_stream_infer("vllm_model", [inp, stream_in, samp_in])
  time.sleep(5)
  client.stop_stream()
  EOF

  Prometheus metrics:
  $ curl http://{{ .Values.inference.serviceName }}:8002/metrics

{{- end }}
